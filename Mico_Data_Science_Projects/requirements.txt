# Dependencies for all three Streamlit apps.
# This file sits at the repo ROOT (Streamlit Cloud installs from the root).
#
# Versions are specified as bounded ranges rather than exact pins so that pip can
# always select a PREBUILT WHEEL on Python 3.11 / 3.12 / 3.13 (Streamlit Cloud's
# default). Exact pins (numpy 1.26, scipy 1.13, matplotlib 3.8, pyarrow 17) have no
# wheels for Python >= 3.13, which forces pip to compile them from source and the
# build hangs at "Preparing metadata (pyproject.toml): still running...".
#
# Constraints kept on purpose:
#   pandas < 3.0   -> seaborn 0.13 boxplots break on pandas >= 3.0
#                     (SeriesGroupBy.grouper was removed)
#   pyarrow >= 18  -> first release with Python 3.13 wheels; requires numpy >= 2.1,
#                     so numpy is also bumped to >= 2.1 (the old pyarrow==17 pin was
#                     only needed to stay on numpy 1.26).
streamlit>=1.41,<1.50
numpy>=2.1,<2.3
pandas>=2.2.3,<3
scipy>=1.14.1,<1.16
scikit-learn>=1.6,<1.8
imbalanced-learn>=0.12.4,<0.15
matplotlib>=3.9.2,<3.11
seaborn>=0.13.2,<0.14
pyarrow>=18,<21
joblib>=1.4,<1.6
plotly>=5.22,<6
openpyxl>=3.1,<4
nbformat>=5.10,<6
